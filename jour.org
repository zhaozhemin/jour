* HTML export backend

I wrote my own export backend for three reasons. First, I want to
fully control the HTML structure. I don't think it's possible to alter
the structure, e.g. which Org element generates which HTML element,
just by customizing some variables. (or maybe I just didn't bother
with reading the docs carefully.)

The second and third reasons are in the same vein. People who created
markup languages like HTML and Org really didn't think it through when
it comes to Asian languages. For example, in HTML, a newline character
is treated like a space. That is natural in English where words are
separated by spaces. Not the case with Chinese, or Japanese. If you
write your text with hard line breaks, good luck when converting to
HTML, because it looks terrible. A particular problem with Org is that
inline markups are also delineated by spaces. Without special
treatment, there will be spaces between normal text and, say, bold
text. One common solution is to use zero-width space. It works but I
don't like it. I like to know what's going on in my text file. If I
see no space, there's no space. If I see a blank, well, is it a space
or is it a tab? I'd like to know! So in a sense, what I'm looking for
in a plaintext file is, wait for it, WYSIWYG. My solution is to remove
the unwanted spaces and newlines during export. The backend will check
the characters at both ends of a space or a newline character and
remove the space if one of the characters is a CJK character.

** Text processing

This section deals with removing spaces around inline markups. It's
pretty self contained. You can plug the function into Org's parse tree
hook, and it would work in Org's own HTML export, I suppose.

#+begin_src emacs-lisp
  (defun jour-remove-space-p (a b)
    "Return true if the last character of `A' or the first character of `B'
  is a CJK character."
    (and
     (stringp a)
     (stringp b)
     (not (string-empty-p a))
     (not (string-empty-p b))
     (or
      (jour-cjk-char-p (substring a -1))
      (jour-cjk-char-p (substring b 0 1)))))

  ;; TODO There're special cases like the link object where it's
  ;; possible to not have children in which case the text should be the
  ;; raw link value.
  (defun jour-get-object-text (object &optional first-or-last)
    "Return the text of an Org object or nil if not found. `OBJECT' 'roughly
  correspond to raw, possibly-formatted text.' See
  `org-element-all-objects'. `FIRST-OR-LAST' is either 'first or 'last,
  specifying which chunk of the text in a recursive object should be
  retrieved."
    (let ((object-type (org-element-type object)))
      (cond
       ((eq object-type 'plain-text) object)
       ((memq object-type '(code verbatim)) (org-element-property :value object))
       ((memq object-type org-element-recursive-objects)
        (let* ((contents (org-element-contents object))
               (maybe-text (if (eq first-or-last 'last) (car (last contents)) (car contents))))
          (if (eq (org-element-type maybe-text) 'plain-text)
              maybe-text
            (jour-get-object-text maybe-text first-or-last))))
       ;; TODO footnote-reference is actually a recursive object. This branch is unreachable.
       ((memq object-type '(footnote-reference)) (org-element-property :label object)))))

  (defun jour-clean-spaces-around-inline-markup (ast backend info)
    "Remove the spaces between the texts and the inline markup object,
  if one of the characters around the space is a CJK character. The
  preceding space is at the end of the previous text; the following
  space is not in the text string but indicated by the :post-blank
  property of the inline markup object."
    (cl-labels ((clean-pre-space (text object)
                  (let* ((ends-with-space-p (string= (substring text -1) " "))
                         (last-char (if ends-with-space-p (substring text -2 -1) (substring text -1))))
                    (cond ((not ends-with-space-p) nil)
                          ((jour-remove-space-p last-char (jour-get-object-text object 'first))
                           (org-element-set-element text (substring text 0 -1))))))

                (clean-post-space (object text-or-object)
                  (cond ((= (org-element-property :post-blank object) 0) nil)
                        ((jour-remove-space-p (jour-get-object-text object 'last)
                                              (jour-get-object-text text-or-object 'first))
                         (org-element-put-property object :post-blank 0))))

                (helper (contents)
                  (cond ((or (null contents) (null (cdr contents))) '())
                        ((and (eq (org-element-type (car contents)) 'plain-text)
                              (memq (org-element-type (cadr contents)) org-element-all-objects))
                         (progn
                           (clean-pre-space (car contents) (cadr contents))
                           (helper (cdr contents))))
                        ((memq (org-element-type (car contents)) org-element-all-objects)
                         (progn
                           (clean-post-space (car contents) (cadr contents))
                           (helper (cdr contents))))
                        (t (helper (cdr contents))))))

      (org-element-map ast '(paragraph)
        (lambda (parent)
          (let ((contents (org-element-contents parent)))
            (helper contents))))

      ast))

  (when (not (member #'jour-clean-spaces-around-inline-markup org-export-filter-parse-tree-functions))
    (add-hook 'org-export-filter-parse-tree-functions #'jour-clean-spaces-around-inline-markup))

  (defvar jour-cjk-char-pat
    (rx (in "\u4E00-\u9FFF"         ; Unified Ideographs
            "\u3400-\u4DBF"         ; Unified Ideographs Extension A
            "\uF900-\uFAFF"         ; Compatibility Ideographs
            "\u3040-\u309F"         ; Hiragana
            "\u30A0-\u30FF"         ; Katakana
            "\u31F0-\u31FF"         ; Katakana Phonetic Extensions
            "\uAC00-\uD7AF"         ; Hangul
            "\u3000-\u303F"         ; Punctuation
            "\uFF00-\uFFEF")))      ; Halfwidth and Fullwidth Forms

  (defun jour-cjk-char-p (string)
    "Check if `STRING' contains any CJK character."
    (string-match-p jour-cjk-char-pat string))
#+end_src

* Post data structure

There are several struct fields which are dependent on other
fields. Normally it's not a good idea to store derived values, since
it can easily get out of sync. That is not an issue here because a
post is never modified except for ~prev-post~ and ~next-post~.

~keywords~ here usually means raw keywords collected from an Org file,
which are parsed and validated into a plist, which is fed into the
constructor.


#+begin_src emacs-lisp
  (cl-defstruct (jour-post (:constructor jour-post--create) (:copier nil))
    (filename nil :documentation "Path of the current post in the filesystem.")
    (date nil :documentation "Elisp time, e.g. time returned by `encode-time'.")
    (slug nil :documentation "Pulled from the source file.")
    (title nil :documentation "Title of the post, nil if it's a micro post.")
    (updated nil :documentation "Update time. Pulled from the source file. Elisp time.")
    (prev-post nil :documentation "Previous post. See `jour-post-nav-data'.")
    (next-post nil :documentation "Next post. See `jour-post-nav-data'.")
    (lang nil :documentation "Language of the post. It will be set on the lang attr on the article element.")
    (publicp nil :documentation "Determine whether a post gets published. Pulled from the source file.")

    (url nil :documentation "Absolute URL for the post, e.g. /2026/02/hi/")
    (output-dir nil :documentation "Relative directory of the output HTML, e.g. build/2026/02/hi/")
    (output-html nil :documentation "Relative filename of the output HTML, e.g. build/2026/02/hi/index.html")
    (microp nil :documentation "Is it a micro post?")
    (date-string nil :documentation "Formatted time string, e.g. 2026.02.02"))

  (defun jour-post-collect-keywords (filename)
    "Collect keywords from an Org file, and add filename as a keyword. Return
  an alist with symbol keys. Repeated keywords are ignored."
    (let* ((keywords '("title" "date" "slug" "updated" "public" "lang"))
           (unique keywords))
      (with-temp-buffer
        (insert-file-contents filename)
        ;; This might slow things down.
        (org-mode)
        (append
         (list (cons 'filename filename))
         (mapcar (lambda (x) (pcase x (`(,k . ,v) (cons (intern (downcase k)) v))))
                 (org-collect-keywords keywords unique))))))

  (defun jour-post-parse-public (val)
    (pcase val ((or "t" "true") t) (_ nil)))

  (defun jour-post-parse-time (time)
    (if (stringp time)
        (condition-case
            nil
            (encode-time (parse-time-string time))
          (wrong-type-argument (user-error "time `%s' is malformatted" time)))
      (user-error "time `%s' is not a string" time)))

  (defun jour-post-parse-keyword (keyword)
    (pcase keyword
      (`(title . ,v) `(:title ,v))
      (`(date . ,v) `(:date ,(jour-post-parse-time v)))
      (`(slug . ,v) `(:slug ,v))
      (`(updated . ,v) `(:updated ,(jour-post-parse-time v)))
      (`(public . ,v) `(:publicp ,(jour-post-parse-public v)))
      (`(lang . ,v) `(:lang ,v))
      (`(filename . ,v) `(:filename ,v))))

  (defun jour-post-parse-keywords (keywords)
    (mapcan #'jour-post-parse-keyword keywords))

  (defun jour-post-check-required-keywords (keywords)
    (dolist (k '(date slug filename))
      (unless (alist-get k keywords)
        (user-error "field `%s' is missing" k))))

  (defun jour-post-build-url (date slug microp)
    (jour-file-name-concat
     / (format-time-string (if microp "%Y/%m/%d" "%Y/%m") date) slug /))

  ;; TODO Org time may not include hours, minutes, and seconds. Maybe
  ;; support multiple time format?
  (defun jour-post-create (keywords)
    (condition-case
        err
        (progn
          (jour-post-check-required-keywords keywords)
          (let* ((metadata (jour-post-parse-keywords keywords))
                 (microp (pcase (plist-get metadata :title)
                           ((or (pred null) (and (pred stringp) (pred string-empty-p))) t)))
                 (url (jour-post-build-url (plist-get metadata :date) (plist-get metadata :slug) microp))
                 (date-string (format-time-string "%Y.%m.%d" (plist-get metadata :date)))
                 (output-dir (jour-file-name-concat jour-output-directory url /))
                 (output-html (jour-file-name-concat output-dir "index.html")))
            (apply #'jour-post--create
                   (append (list :microp microp
                                 :url url
                                 :date-string date-string
                                 :output-dir output-dir
                                 :output-html output-html)
                           metadata))))
      (error
       (user-error "%s: %s"
                   (file-name-nondirectory (alist-get 'filename keywords))
                   (error-message-string err)))))

  (defun jour-get-post-title (post)
    (if (jour-post-microp post)
        (format "Posted on %s" (jour-post-date-string post))
      (jour-post-title post)))

  (defun jour-post-nav-data (post)
    "Extract url and title for inter post navigation use. Return an
  alist with symbol keys."
    (list (cons 'url (jour-post-url post))
          (cons 'title (jour-get-post-title post))))
#+end_src

    (when (not (null (dom-tag dom)))
      (insert (format "<%s" (dom-tag dom)))
      (dolist (attr (dom-attributes dom))
        (insert (format " %s=\"%s\"" (car attr) (cdr attr))))
      (insert ">")
      (cond
       ((memq (dom-tag dom) jour-html-void-elements))
       (t
        (dolist (elem (dom-children dom))
          (cond
           ((stringp elem) (insert elem))
           (t (jour-dom-print elem))))
        (insert (format "</%s>" (dom-tag dom)))))))

  (defun jour-dom-to-html (dom &optional doctype)
    (with-temp-buffer
      (when doctype
        (insert "<!doctype html>\n"))
      (jour-dom-print dom)
      (buffer-string)))

  (defun jour-get-posts ()
    "Return public posts in descending date order."
    (let* ((filenames (directory-files jour-posts-directory t "^[^#\\.].+\\.org$"))
           (entries (mapcar #'jour-post-collect-keywords filenames))
           (public-entries (seq-filter (lambda (entry) (jour-post-parse-public (alist-get 'public entry))) entries))
           (posts (mapcar #'jour-post-create public-entries)))
      (sort posts (lambda (a b) (time-less-p (jour-post-date b) (jour-post-date a))))))

  (defun jour-posts-add-neighbors (sorted-posts)
    "Add previous and next post data in-place. Elisp seems to be very
  permissive in this regard that car/cdr like operations on an
  empty list all return nil."
    (cl-labels ((helper (current rest prev-post next-post)
                  (cond ((eq current nil) nil)
                        (t (ignore-error wrong-type-argument
                             (setf (jour-post-prev-post current) (jour-post-nav-data prev-post)))
                           (ignore-error wrong-type-argument
                             (setf (jour-post-next-post current) (jour-post-nav-data next-post)))
                           (helper (car rest) (cdr rest) (cadr rest) current)))))

      (helper (car sorted-posts) (cdr sorted-posts) (cadr sorted-posts) '())))

  (defun jour-build-output-directory ()
    "Create the directory for output files. Tangle assets source
  blocks, e.g. CSS and images, into files."
    (when (file-directory-p jour-output-directory)
      (move-file-to-trash jour-output-directory))
    (make-directory jour-assets-directory t)
    (org-babel-tangle-file current-file)
    (jour-cache-busting-css))

  (defun jour-build ()
    (interactive)
    (with-temp-buffer
      (setq default-directory (file-name-directory current-file))
      (let ((posts (jour-get-posts)))
        (jour-build-output-directory)
        (jour-posts-add-neighbors posts)
        (mapc 'jour-build-post-page posts)
        (jour-build-main-index (take 10 posts) (nth 10 posts))
        (jour-build-rss (take 10 posts))
        (jour-build-quotes-page)
        (jour-build-hypertexts-page)
        (jour-build-archive-page (seq-filter (lambda (p) (not (jour-post-microp p))) posts))
        (jour-build-micro-post-archive-page (seq-filter (lambda (p) (jour-post-microp p)) posts))
        (jour-build-about-page))))

  (defun jour--file-name-concat-helper (rootp dirp &rest names)
    (cl-flet ((helper (names0)
                (string-join (mapcan (lambda (s) (string-split s "/" t)) names0) "/")))
      (concat (if rootp "/" "")
              (helper names)
              (if dirp "/" ""))))

    "Join file names, removing unnecessary slashes, adding leading or
  ending slash."
    (let ((rootp (eq (car names) '/))
          (dirp (eq (car (last names)) '/))
          (names0 (remq '/ names)))
      `(jour--file-name-concat-helper ,rootp ,dirp ,@names0)))
